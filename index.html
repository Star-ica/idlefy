Smart Contract

Each wallet has its own to-do list stored onchain. Tasks are scoped to msg.sender.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/// @title OnchainTodoList
/// @notice Each address owns and manages their own list of tasks
contract OnchainTodoList {

    struct Task {
        uint256 id;
        string  text;
        bool    completed;
        uint256 createdAt;
        uint256 completedAt;
    }

    // owner => tasks
    mapping(address => Task[]) private _tasks;
    mapping(address => uint256) private _nextId;

    event TaskAdded(address indexed owner, uint256 id, string text);
    event TaskCompleted(address indexed owner, uint256 id);
    event TaskDeleted(address indexed owner, uint256 id);

    function addTask(string calldata text) external {
        require(bytes(text).length > 0, "Empty task");
        uint256 id = _nextId[msg.sender]++;
        _tasks[msg.sender].push(Task({
            id: id,
            text: text,
            completed: false,
            createdAt: block.timestamp,
            completedAt: 0
        }));
        emit TaskAdded(msg.sender, id, text);
    }

    function completeTask(uint256 taskIndex) external {
        Task storage t = _tasks[msg.sender][taskIndex];
        require(!t.completed, "Already done");
        t.completed = true;
        t.completedAt = block.timestamp;
        emit TaskCompleted(msg.sender, t.id);
    }

    function deleteTask(uint256 taskIndex) external {
        Task[] storage list = _tasks[msg.sender];
        require(taskIndex < list.length, "Out of range");
        uint256 id = list[taskIndex].id;
        list[taskIndex] = list[list.length - 1];
        list.pop();
        emit TaskDeleted(msg.sender, id);
    }

    function getTasks()
        external view returns (Task[] memory)
    {
        return _tasks[msg.sender];
    }

    function getTaskCount() external view returns (uint256) {
        return _tasks[msg.sender].length;
    }

    function getPendingCount() external view returns (uint256 count) {
        Task[] memory list = _tasks[msg.sender];
        for (uint256 i; i < list.length; i++) {
            if (!list[i].completed) count++;
        }
    }
}
